module ActivationFuncs;


export public void LeakyReLU <let N : int>(inout CoopVec<half, N> layer)
{
    // [ForceUnroll]
    // for (int i = 0; i < N; i++)
    // {
    //     layer[i] *= half(layer[i] > 0 ? 1 : 0.1);
    // }

    layer = max(layer, CoopVec<half, N>(half(0.))) + min(layer, CoopVec<half, N>(half(0.))) * half(0.1);
}

export public void Softmax<let N : int>(inout CoopVec<half, N> layer)
{
    float16_t expSum  = 0;
    float16_t largest = 0;

    for (int i = 0; i < layer.getCount(); i++)
    {
        largest = (layer[i] > largest ? layer[i] : largest);
    }

    for (int i = 0; i < layer.getCount(); i++)
    {
        layer[i] -= largest;
    }
    layer = exp(layer);

    for (int i = 0; i < 10; i++)
    {
        expSum += layer[i];
    }

    for (int i = 0; i < 10; i++)
        layer[i] /= expSum;
}
