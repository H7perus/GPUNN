import ActivationFuncs;

struct MnistDatapoint
{
    float16_t img[28 * 28];
    uint16_t  label;
}


#define INPUT_FEATURES 28 * 28

#define HIDDEN_NEURONS 64
#define HIDDEN_LAYERS 3
// The output is vec<float, 10>
#define OUTPUT_NEURONS 10

[vk::binding(0, 0)]
RWByteAddressBuffer input; // [1 2 3 4 …]
[vk::binding(0, 0)]
RWStructuredBuffer<MnistDatapoint> mnistData;

[vk::binding(1, 0)]
RWStructuredBuffer<float16_t> outputBuffer;

[vk::binding(2, 0)]
RWByteAddressBuffer weights;
[vk::binding(2, 0)]
RWStructuredBuffer<float16_t> weightsFP16;

[vk::binding(3, 0)]
RWByteAddressBuffer weightAdjustment;
[vk::binding(3, 0)]
RWStructuredBuffer<float16_t> weightAdjustmentFP16;


struct PushConstants
{
    int imageBase;
    int isTraining;
    int totalSize;
}

[[vk::push_constant]]
PushConstants pc;

[shader("compute")]
[numthreads(64, 1, 1)]
void computeMain(uint3 threadID: SV_DispatchThreadID)
{
    int imageIndex = threadID.x + pc.imageBase;

    if (imageIndex >= 60000 + 10000 * (1 - pc.isTraining))
        return;

    CoopVec<float16_t, INPUT_FEATURES> vec =
        coopVecLoad<INPUT_FEATURES, float16_t>(input, sizeof(MnistDatapoint) * imageIndex);


    int buffOffset = 0;

    CoopVec<float16_t, HIDDEN_NEURONS> intermediateVecs[HIDDEN_LAYERS];

    intermediateVecs[0] =
        coopVecMatMul<float16_t, HIDDEN_NEURONS, INPUT_FEATURES>(vec,                           // input vector
                                                                 CoopVecComponentType::Float16, // vector interpretation
                                                                 weights,                       // input matrix
                                                                 0,                             // matrix offset in byte
                                                                 CoopVecComponentType::Float16, // matrix interpretation
                                                                 CoopVecMatrixLayout::TrainingOptimal, // matrix layout
                                                                 false,             // matrix transpose
                                                                 INPUT_FEATURES * 2 // matrix stride
        );

    //Moving to after  the first matrix (input * hidden sized, times two for the float16)
    buffOffset += INPUT_FEATURES * HIDDEN_NEURONS * 2;

    LeakyReLU(intermediateVecs[0]);

    for (int i = 1; i < HIDDEN_LAYERS; i++)
    {
        intermediateVecs[i] = coopVecMatMul<float16_t, HIDDEN_NEURONS, HIDDEN_NEURONS>(
            intermediateVecs[i-1],                                  // input vector
            CoopVecComponentType::Float16,        // vector interpretation
            weights,                              // input matrix
            buffOffset,                           // matrix offset in byte
            CoopVecComponentType::Float16,        // matrix interpretation
            CoopVecMatrixLayout::TrainingOptimal, // matrix layout
            false,                                // matrix transpose
            HIDDEN_NEURONS * 2                    // matrix stride
        );

        buffOffset += HIDDEN_NEURONS * HIDDEN_NEURONS * 2;
        LeakyReLU(intermediateVecs[i]);
        
    }


    CoopVec<float16_t, OUTPUT_NEURONS> result;
    result =
        coopVecMatMul<float16_t, OUTPUT_NEURONS, HIDDEN_NEURONS>(intermediateVecs[HIDDEN_LAYERS - 1],
                                                                 CoopVecComponentType::Float16, // vector interpretation
                                                                 weights,                       // input matrix
                                                                 buffOffset,                    // matrix offset in byte
                                                                 CoopVecComponentType::Float16, // matrix interpretation
                                                                 CoopVecMatrixLayout::TrainingOptimal, // matrix layout
                                                                 false,             // matrix transpose
                                                                 HIDDEN_NEURONS * 2 // matrix stride
        );

    Softmax(result);

    int outputOffset = imageIndex * OUTPUT_NEURONS;
    for (int i = 0; i < OUTPUT_NEURONS && !bool(pc.isTraining); i++)
    {
        outputBuffer[i + outputOffset] = result[i]; // [30 70 110 150 …]
    }

    if (!bool(pc.isTraining))
        return;


    CoopVec<float16_t, OUTPUT_NEURONS> outputGrad;

    for (int i = 0; i < OUTPUT_NEURONS; i++)
    {
        outputGrad[i] = ((mnistData[imageIndex].label == i ? 1 : 0) - result[i]);
    }
    if (pc.isTraining == 1)
    {
        coopVecOuterProductAccumulate(outputGrad, intermediateVecs[HIDDEN_LAYERS - 1], weightAdjustment, buffOffset,
                                      128, CoopVecMatrixLayout::TrainingOptimal, CoopVecComponentType::Float16);

        CoopVec<float16_t, HIDDEN_NEURONS> intermediateGrad = coopVecMatMul<float16_t, HIDDEN_NEURONS, OUTPUT_NEURONS>(
            outputGrad,
            CoopVecComponentType::Float16,        // vector interpretation
            weights,                              // input matrix
            buffOffset,                           // matrix offset in byte
            CoopVecComponentType::Float16,        // matrix interpretation
            CoopVecMatrixLayout::TrainingOptimal, // matrix layout
            true,                                 // matrix transpose
            HIDDEN_NEURONS * 2                    // matrix stride
        );
        for (int j = 0; j < HIDDEN_NEURONS; j++)
        {
            float factor        = (intermediateVecs[HIDDEN_LAYERS -1][j] < 0) ? 0.1f : 1.0f;
            intermediateGrad[j] = intermediateGrad[j] * factor;
        }

        for (int i = HIDDEN_LAYERS - 2; i >= 0; i--)
        {
            buffOffset -= HIDDEN_NEURONS * HIDDEN_NEURONS * 2;
            coopVecOuterProductAccumulate(intermediateGrad, intermediateVecs[i], weightAdjustment, buffOffset, 128,
                                          CoopVecMatrixLayout::TrainingOptimal, CoopVecComponentType::Float16);


            intermediateGrad = coopVecMatMul<float16_t, HIDDEN_NEURONS, HIDDEN_NEURONS>(
                intermediateGrad,
                CoopVecComponentType::Float16,        // vector interpretation
                weights,                              // input matrix
                buffOffset,                           // matrix offset in byte
                CoopVecComponentType::Float16,        // matrix interpretation
                CoopVecMatrixLayout::TrainingOptimal, // matrix layout
                true,                                 // matrix transpose
                HIDDEN_NEURONS * 2                    // matrix stride
            );

            for (int j = 0; j < HIDDEN_NEURONS; j++)
            {
                float factor        = (intermediateVecs[i][j] < 0) ? 0.1f : 1.0f;
                intermediateGrad[j] = intermediateGrad[j] * factor;
            }
        }
        buffOffset = 0;
        coopVecOuterProductAccumulate(intermediateGrad, vec, weightAdjustment, buffOffset, INPUT_FEATURES * 2,
                                      CoopVecMatrixLayout::TrainingOptimal, CoopVecComponentType::Float16);
    }
}
